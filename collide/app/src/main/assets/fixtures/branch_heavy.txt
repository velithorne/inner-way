fun classifyScore(score: Int): String {
    return when {
        score >= 90 -> "A"
        score >= 80 -> "B"
        score >= 70 -> "C"
        score >= 60 -> "D"
        else -> "F"
    }
}

fun determineShipping(weight: Float, destination: String, expedited: Boolean): Float {
    val baseRate = if (destination == "local") {
        if (weight < 1.0f) 2.99f else 4.99f
    } else if (destination == "national") {
        if (weight < 1.0f) 5.99f else 9.99f
    } else {
        if (weight < 1.0f) 12.99f else 24.99f
    }
    return if (expedited) baseRate * 2.5f else baseRate
}

fun categorizeAge(age: Int): String {
    return if (age < 0) {
        "invalid"
    } else if (age < 13) {
        "child"
    } else if (age < 18) {
        "teenager"
    } else if (age < 65) {
        "adult"
    } else {
        "senior"
    }
}
